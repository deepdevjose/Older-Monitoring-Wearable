  {\textbf{No active episode}\\No active alert pointer\\Per patient and rule};

  {\textbf{Open episode}\\One persistent alert ID\\Duplicates suppressed};

  {\textbf{Recovered episode}\\\texttt{recovered\_at} recorded\\Active rule pointer cleared};
\draw[transition] (idle) -- node[edge label,above,text width=1.8cm]{3 samples:\\rule triggered} (open);
\draw[transition] (open) -- node[edge label,above,text width=1.8cm]{3 samples:\\rule absent} (recovered);
\draw[transition] (recovered.north) -- ++(0,1.3) -|
  node[edge label,pos=0.48,above]{New three-sample trigger: create a new alert ID} (open.north);
\draw[transition] (open.south) -- (7.25,-1.55) -- (5.55,-1.55) -- (open.south west);
\node[edge label,text width=3.1cm] at (3.65,-1.65)
  {Continued trigger or\\ineligible input:\\keep episode open};
\node[draw=warning!65,fill=warning!6,rounded corners=3pt,
  text width=5.3cm,minimum height=1.05cm,align=center,inner sep=6pt] (review) at (11.15,-2.25)
  {\textbf{Independent caregiver review}\\Insert \texttt{(alert\_id, user\_id)} acknowledgement.\\Does not close the episode or certify recovery.};
\draw[->,dashed,draw=warning] (open.south east) -- (review.north west);
\node[draw=none,fill=light,rounded corners=3pt,text width=14.3cm,
  align=left,inner sep=8pt] at (7.25,-4.40)
  {\textbf{Eligibility is a gate, not another clinical state.}
   The latest three readings must be synthetic, fresh within the supported two-minute window,
   ordered as the latest capture, and separated by 1--30 seconds. Rule-specific validity/contact
   checks apply. Recovery requires valid non-triggering observations of that rule;
   stale, missing, invalid or device-diagnostic input cannot establish recovery.};
\end{tikzpicture}}
\caption{Implemented educational episode lifecycle for one patient--rule pair, derived from \texttt{alerts.py} and the SQLite schema. The recovered alert remains in history while its active pointer is cleared. ``Recovered'' describes the synthetic rule episode; it is not a health outcome. Caregiver review is an independent per-user record.}
\label{fig:alert-state}
\end{figure}%
}
