// =====================================================
// I2C HELPER
// =====================================================

bool i2cDeviceExists(uint8_t address)
{
  Wire.beginTransmission(address);

  return Wire.endTransmission() == 0;
}


// =====================================================
// READ I2C BYTES
// =====================================================

bool readBytes(
  uint8_t address,
  uint8_t reg,
  uint8_t *buffer,
  uint8_t length
)
{
  Wire.beginTransmission(address);

  Wire.write(reg);

  if (Wire.endTransmission(false) != 0)
  {
    return false;
  }

  uint8_t received =
    Wire.requestFrom(address, length, true);

  if (received != length)
  {
    return false;
  }

  for (uint8_t i = 0; i < length; i++)
  {
    buffer[i] = Wire.read();
  }

  return true;
}


// =====================================================
// WRITE I2C BYTE
// =====================================================

bool writeByte(
  uint8_t address,
  uint8_t reg,
  uint8_t value
)
{
  Wire.beginTransmission(address);

  Wire.write(reg);
  Wire.write(value);

  return Wire.endTransmission() == 0;
}


// =====================================================
// INITIALIZE MPU6050
// =====================================================

void initMPU()
{
  if (!i2cDeviceExists(MPU6050_ADDRESS))
  {
    Serial.println("MPU6050 not found");

    mpuOnline = false;

    return;
  }

  // Wake MPU6050
  if (!writeByte(
        MPU6050_ADDRESS,
        0x6B,
        0x00
      ))
  {
    mpuOnline = false;

    return;
  }

  mpuOnline = true;

  Serial.println("MPU6050 found");
}


// =====================================================
// UPDATE MPU6050
// =====================================================

void updateMPU()
{
  if (!mpuOnline)
  {
    return;
  }

  uint8_t data[14];

  if (!readBytes(
        MPU6050_ADDRESS,
        0x3B,
        data,
        14
      ))
  {
    mpuOnline = false;

    return;
  }

  int16_t rawAX =
    (data[0] << 8) | data[1];

  int16_t rawAY =
    (data[2] << 8) | data[3];

  int16_t rawAZ =
    (data[4] << 8) | data[5];


  int16_t rawGX =
    (data[8] << 8) | data[9];

  int16_t rawGY =
    (data[10] << 8) | data[11];

  int16_t rawGZ =
    (data[12] << 8) | data[13];


  accelX = rawAX / 16384.0;
  accelY = rawAY / 16384.0;
  accelZ = rawAZ / 16384.0;


  gyroX = rawGX / 131.0;
  gyroY = rawGY / 131.0;
  gyroZ = rawGZ / 131.0;
}


// =====================================================
// READ MLX90614 TEMPERATURE
// =====================================================

float readMLXTemperature(uint8_t reg)
{
  Wire.beginTransmission(GY906_ADDRESS);

  Wire.write(reg);

  if (Wire.endTransmission(false) != 0)
  {
    return NAN;
  }

  Wire.requestFrom(
    GY906_ADDRESS,
    (uint8_t)3
  );

  if (Wire.available() < 3)
  {
    return NAN;
  }

  uint8_t low = Wire.read();
  uint8_t high = Wire.read();

  // PEC byte
  Wire.read();

  uint16_t raw =
    ((uint16_t)high << 8) | low;

  float temperature =
    raw * 0.02 - 273.15;

  return temperature;
}

// =====================================================
// INITIALIZE MAX30102
// =====================================================

void initMAX30102()
{
  Serial.println("Initializing MAX30102...");


  maxOnline = optical.begin(
    Wire,
    I2C_SPEED_STANDARD,
    MAX30102_ADDRESS
  );


  if (!maxOnline)
  {
    Serial.println("MAX30102 not found");

    maxStatus = "not_found";

    return;
  }


  Serial.println("MAX30102 found");


  /*
      LED brightness = 60
      Sample average = 4
      LED mode = 2 (Red + IR)
      Sample rate = 100
      Pulse width = 411
      ADC range = 4096
  */

  optical.setup(
    60,
    4,
    2,
    100,
    411,
    4096
  );


  optical.setPulseAmplitudeRed(0x1F);

  optical.setPulseAmplitudeIR(0x1F);

  optical.setPulseAmplitudeGreen(0);


  ppgCount = 0;

  maxStatus = "waiting";
}


// =====================================================
// UPDATE MAX30102
// =====================================================

void updateMAX30102()
{
  if (!maxOnline)
  {
    return;
  }


  /*
      Pull new samples from the MAX30102 hardware FIFO.

      IMPORTANT:
      We do NOT manually reset the FIFO anymore.
  */

  optical.check();


  while (optical.available())
  {

    uint32_t red =
      optical.getFIFORed();

    uint32_t ir =
      optical.getFIFOIR();


    optical.nextSample();


    latestRed = red;
    latestIR = ir;


    // -------------------------------------------------
    // Finger detection
    // -------------------------------------------------

    fingerDetected =
      (ir > FINGER_THRESHOLD);


    if (!fingerDetected)
    {
      maxStatus = "no_finger";

      hrValid = false;
      spo2Valid = false;

      heartRate = 0;
      spo2 = 0;

      ppgCount = 0;

      continue;
    }


    // -------------------------------------------------
    // Finger detected
    // -------------------------------------------------

    maxStatus = "warming_up";


    // Store sample

    irBuffer[ppgCount] = ir;

    redBuffer[ppgCount] = red;

    ppgCount++;


    // -------------------------------------------------
    // Wait until 100 samples collected
    // -------------------------------------------------

    if (ppgCount >= PPG_SAMPLES)
    {

      int8_t spo2ValidFlag = 0;
      int8_t heartRateValidFlag = 0;


      maxim_heart_rate_and_oxygen_saturation(
        irBuffer,
        PPG_SAMPLES,
        redBuffer,
        &spo2,
        &spo2ValidFlag,
        &heartRate,
        &heartRateValidFlag
      );


      // -------------------------------------------------
      // Validate heart rate
      // -------------------------------------------------

      hrValid =
        heartRateValidFlag &&
        heartRate >= 30 &&
        heartRate <= 220;


      // -------------------------------------------------
      // Validate SpO2
      // -------------------------------------------------

      spo2Valid =
        spo2ValidFlag &&
        spo2 >= 70 &&
        spo2 <= 100;


      // -------------------------------------------------
      // Sensor status
      // -------------------------------------------------

      if (hrValid && spo2Valid)
      {
        maxStatus = "ok";
      }
      else if (hrValid || spo2Valid)
      {
        maxStatus = "partial";
      }
      else
      {
        maxStatus = "poor_signal";
      }


      // -------------------------------------------------
      // Keep newest 75 samples
      // -------------------------------------------------

      memmove(
        irBuffer,
        irBuffer + PPG_STEP,
        (PPG_SAMPLES - PPG_STEP)
          * sizeof(irBuffer[0])
      );


      memmove(
        redBuffer,
        redBuffer + PPG_STEP,
        (PPG_SAMPLES - PPG_STEP)
          * sizeof(redBuffer[0])
      );


      ppgCount =
        PPG_SAMPLES - PPG_STEP;
    }
  }
}

// =====================================================
// PRINT JSON
// =====================================================

void printJSON()
{
  JsonDocument doc;


  // -------------------------------------------------
  // Device
  // -------------------------------------------------

  doc["device_id"] = DEVICE_ID;

  doc["uptime_ms"] = millis();


  // -------------------------------------------------
  // MAX30102
  // -------------------------------------------------

  JsonObject max =
    doc["max30102"].to<JsonObject>();


  max["i2c_address"] = "0x57";

  max["status"] = maxStatus;

  max["finger_detected"] =
    fingerDetected;


  max["heart_rate_valid"] =
    hrValid;

  max["spo2_valid"] =
    spo2Valid;


  if (hrValid)
  {
    max["heart_rate_bpm"] =
      heartRate;
  }
  else
  {
    max["heart_rate_bpm"] =
      nullptr;
  }


  if (spo2Valid)
  {
    max["spo2_percent"] =
      spo2;
  }
  else
  {
    max["spo2_percent"] =
      nullptr;
  }


  max["ir_raw"] =
    latestIR;

  max["red_raw"] =
    latestRed;


  max["window_samples"] =
    ppgCount;

  max["window_target"] =
    PPG_SAMPLES;


  // -------------------------------------------------
  // MPU6050
  // -------------------------------------------------

  JsonObject mpu =
    doc["mpu6050"].to<JsonObject>();


  mpu["i2c_address"] = "0x68";

  mpu["online"] =
    mpuOnline;


  JsonObject accel =
    mpu["accel_g"].to<JsonObject>();


  accel["x"] = accelX;
  accel["y"] = accelY;
  accel["z"] = accelZ;


  JsonObject gyro =
    mpu["gyro_dps"].to<JsonObject>();


  gyro["x"] = gyroX;
  gyro["y"] = gyroY;
  gyro["z"] = gyroZ;


  // -------------------------------------------------
  // GY906
  // -------------------------------------------------

  JsonObject temp =
    doc["gy906"].to<JsonObject>();


  temp["i2c_address"] = "0x5A";

  temp["online"] =
    gyOnline;


  if (
    gyOnline &&
    !isnan(objectTemp)
  )
  {
    temp["object_temp_c"] =
      objectTemp;
  }
  else
  {
    temp["object_temp_c"] =
      nullptr;
  }


  if (
    gyOnline &&
    !isnan(ambientTemp)
  )
  {
    temp["ambient_temp_c"] =
      ambientTemp;
  }
  else
  {
    temp["ambient_temp_c"] =
      nullptr;
  }


  // -------------------------------------------------
  // Microphone
  // -------------------------------------------------

  JsonObject microphone =
    doc["microphone"].to<JsonObject>();


  microphone["pin"] =
    MIC_PIN;

  microphone["adc_raw"] =
    micValue;


  // -------------------------------------------------
  // Photoresistor
  // -------------------------------------------------

  JsonObject photo =
    doc["photoresistor"].to<JsonObject>();


  photo["pin"] =
    PHOTO_PIN;

  photo["adc_raw"] =
    photoValue;


  photo["percent"] =
    photoValue * 100.0 / 4095.0;


  // -------------------------------------------------
  // Buzzer
  // -------------------------------------------------

  JsonObject buzzer =
    doc["buzzer"].to<JsonObject>();


  buzzer["pin"] =
    BUZZER_PIN;


  // -------------------------------------------------
  // Print JSON
  // -------------------------------------------------

  String json;
  serializeJson(doc, json);
  Serial.println(json);

  if (WiFi.status() != WL_CONNECTED)
  {
    Serial.println("HTTP skipped: Wi-Fi is not connected.");
    return;
  }

  HTTPClient http;
  http.begin(API_URL);
  http.addHeader("Content-Type", "application/json");
  http.addHeader("X-API-Key", API_KEY);
  http.setTimeout(HTTP_TIMEOUT);

  int statusCode = -1;
  for (int attempt = 0; attempt < 2; attempt++)
  {
    statusCode = http.POST(json);
    if (statusCode > 0)
    {
      break;
    }
    if (attempt == 0)
    {
      delay(HTTP_RETRY_DELAY);
    }
  }

  if (statusCode > 0)
  {
    Serial.printf("POST /sensor-data -> HTTP %d\n", statusCode);
    if (statusCode == 401 || statusCode == 403)
    {
      Serial.println("Authentication failed: check API_KEY.");
    }
  }
  else
  {
    Serial.printf("POST /sensor-data failed: %s\n",
                  http.errorToString(statusCode).c_str());
  }

  http.end();
}


// =====================================================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(115200);

  delay(1000);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  Serial.println();
  Serial.println("Starting system...");
  Serial.printf("Connecting to hotspot: %s\n", WIFI_SSID);


  // -------------------------------------------------
  // Pins
  // -------------------------------------------------

  pinMode(
    BUZZER_PIN,
    OUTPUT
  );

  digitalWrite(
    BUZZER_PIN,
    LOW
  );


  pinMode(
    MIC_PIN,
    INPUT
  );

  pinMode(
    PHOTO_PIN,
    INPUT
  );


  analogReadResolution(12);


  // -------------------------------------------------
  // I2C
  // -------------------------------------------------

  Wire.begin(
    SDA_PIN,
    SCL_PIN
  );


  Wire.setClock(100000);


  // -------------------------------------------------
  // Sensors
  // -------------------------------------------------

  initMPU();

  initGY906();

  initMAX30102();


  Serial.println();
  Serial.println(
    "System initialized."
  );

  Serial.println("Wi-Fi connection is handled in the background.");
  Serial.println(
    "Place finger steadily on MAX30102."
  );

  Serial.println();
}


// =====================================================
// LOOP
// =====================================================

void loop()
{
  if (WiFi.status() == WL_CONNECTED)
  {
    static bool ipPrinted = false;
    if (!ipPrinted)
    {
      ipPrinted = true;
      Serial.print("Wi-Fi connected. ESP32 IP: ");
      Serial.println(WiFi.localIP());
      Serial.print("FastAPI target: ");
      Serial.println(API_URL);
    }
  }
  else if (millis() - lastWiFiRetry >= WIFI_RETRY_INTERVAL)
  {
    lastWiFiRetry = millis();
    WiFi.disconnect();
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    Serial.println("Retrying Wi-Fi connection...");
  }

  /*
      MAX30102 gets highest priority.

      This function must run very frequently.
  */

  updateMAX30102();


  // Other sensors

  static unsigned long lastMPU = 0;

  if (
    millis() - lastMPU >= 50
  )
  {
    lastMPU = millis();

    updateMPU();
  }


  static unsigned long lastTemp = 0;

  if (
    millis() - lastTemp >= 1000
  )
  {
    lastTemp = millis();

    updateGY906();
  }


  static unsigned long lastAnalog = 0;

  if (
    millis() - lastAnalog >= 100
  )
  {
    lastAnalog = millis();

    updateMicrophone();

    updatePhotoresistor();
  }


  // -------------------------------------------------
  // JSON every second
  // -------------------------------------------------

  if (
    millis() - lastReport
      >= REPORT_INTERVAL
  )
  {
    lastReport = millis();

    printJSON();
  }


  // Let ESP32 background tasks run

  delay(1);
}
